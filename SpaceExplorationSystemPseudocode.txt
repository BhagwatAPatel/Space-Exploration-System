BEGIN CLASS MissionController
    numOfMissions = 0

    METHOD: storeInArray
    IMPORT: pDataFile (String)
    EXPORT: missions (Mission[])
    ASSERTION: Read data from CSV file and stroe in an array of Mission objects
    ALGORITHM:
        FILEINPUTSTREAM fileStream = NULL
        INPUTSTREAMREADER reader 
        BUFFEREDREADER bufferedReader
        STRING line
        MISSION[] missions = NEW Mission[100]

        fileStream = NEW FileInputStream(pDataFile)
        reader = NEW InputStreamReader(fileStream)
        bufferedReader = NEW BufferedReader(reader)
        numOfMissions = 0 
        line = bufferedReader.readLine()

        WHILE line = bufferedReader.readLine() NOT EQUAL NULL DO 
            String[] missionParts = line SPLIT AT "," 
            ASTRONAUT[] missionAstronauts = NEW Astronaut[0]

            IF missionParts[6] IS EMPTY THEN 
                missionAstronauts = NEW Astronaut[0]
            ELSE 
                String[] numberOfAstronauts = missionParts[6] SPLIT AT "|"
                missionAstronauts = new Astronaut[numberOfAstronauts.length]

                FOR i = 0 TO numberOfAstronauts.length CHANGEBY i++
                    STRING[] astronautParts = numberOfAstronauts[i] SPLIT AT ":"
                    missionAstronauts[i] = NEW Astronaut(astronautParts[0], astronaut[1], PARSE INTEGER astronautParts[2], astronautParts[3])
                END FOR 
            END IF 

            missions[numOfMissions++] = NEW Mission(missionParts[0], missionParts[1], missionParts[2], PARSE INTERGER missionParts[3], PARSE DOUBLE missionParts[4], PARSE BOOLEAN missionParts[5], missionAstronauts)
        END WHILE
        CLOSE fileStream

        RETURN missions 
    END storeInArray

    METHOD: readFile 
    IMPORT: pDataFile (String)
    EXPORT: none 
    ASSERTION: reads given file 
    ALGORITHM: 
        FILEINPUTSTREAM fileStream = NULL
        INPUTSTREAMREADER reader 
        BUFFEREDREADER bufferedReader
        STRING line

        fileStream = NEW FileInputStream(pDataFile)
        reader = NEW InputStreamReader(fileStream)
        bufferedReader = NEW BufferedReader(reader)
        line = bufferedReader.readLine()

        WHILE line NOT EQUAL NULL DO 
            PRINT line 
            line = bufferedReader.readLine()
        END WHILE 
        CLOSE fileStream 
    END readFile 

    METHOD: mannedMission
    IMPORT: missions (Mission[])
    EXPORT: none
    ASSERTION: Prints all manned missions 
    ALGORITHM: 
        FOR i = 0 TO missions.length CHANGEBY i++ 
            IF missions[i] NOT EQUAL NULL 
                IF missions[i].getMissionManned
                    MISSION mission = missions[i]
                    PRINT missions[i] + "\n" 
                    IF mission.getAstronaut NOT EQUAL NULL 
                        PRINT "Astronauts part of " + missions[i].getMissionName + ":"
                        FOR ASTRONAUT astronaut : mission.getAstronaut
                            IF astronaut NOT EQUAL NULL
                                PRINT astronaut + "\n"
                            END IF 
                        END FOR 
                    END IF 
                END IF 
            END IF 
        END FOR 
    END mannedMission

    METHOD: unmannedMission 
    IMPORT: missions (Mission[])
    EXPORT: none 
    ASSERTION: Prints all unmanned missions 
    ALGORITHM: 
        FOR i = 0 TO missions.length CHANGEBY i++
            IF missions[i] NOT EQUAL NULL 
                IF NOT missions[i].getMissionManned
                    PRINT missions[i] + "\n"
                END IF 
            END IF 
        END FOR 
    END unmannedMission

    METHOD: findAstronauts
    IMPORT: missions (Mission[])
    EXPORT: none 
    ASSERTION: Prints the astronauts from a given mission
    ALGORITHM: 
        missionAstronaut <- ASK "Which mission's Astronaut would you like to view? "

        FOR i = 0 TO missions.length CHANGEBY i++
            MISSION mission = missions[i]
            IF mission NOT EQUAL NULL 
                IF mission.getMissionCode OR mission.getMissionName EQUAL missionAstronaut
                    IF mission.getMissionManned AND mission.getAstronaut NOT EQUAL NULL
                        PRINT "Astronauts in " + mission.getMissionName + ":"

                        FOR ASTRONAUT astronauts : mission.getAstronaut
                            IF astronauts NOT EQUAL NULL 
                                PRINT astronauts + "\n"
                            END IF 
                        END FOR 
                    ELSE 
                        PRINT "\nMission is unmanned.\n"
                    END IF 
                END IF 
            END IF 
        END FOR 
    END findAstronauts

    METHOD: missionObject 
    IMPORT: none 
    EXPORT: adventures (Mission)
    ASSERTION: Creates a mission object
    ALGORITHM: 
        missionName <- ASK "Enter mission name: "
        missionCode <- ASK "Enter mission code: "
        destinationPlanet <- ASK "Enter destination planet: " 
        launchYear <- ASK "Enter launch year: " 
        successRate <- ASK "Enter success rate: "
        missionManned <- ASK "Is the mission manned? " 

        MISSION adventures = NEW Mission(missionName, missionCode, destinationPlanet, launchYear, successRate, isManned(missionManned))

        RETURN adventures
    END missionObject 

    METHOD: isManned 
    IMPORT: isManned (Boolean)
    EXPORT: manned (Boolean)
    ASSERTION: Determines whether mission is manned or unmanned 
    ALGORITHM: 
        IF isManned EQUALS "yes" OR isManned EQUALS "true" 
            manned = true 
            RETURN manned
        ELSE 
            manned = false 
            RETURN manned 
        END IF 
    END isManned 

    METHOD: astronautArray 
    IMPORT: none
    EXPORT: astronauts (Astronaut[])
    ASSERTION: Creates a new Astronaut array 
    ALGORITHM:
        numberOfAstronauts <- ASK "How many astronauts are part of this mission? " 

        ASTRONAUT[] astronauts = NEW Astronaut[numberOfAstronauts]

        FOR i = 0 TO numberOfAstronauts CHANGEBY i++ 
            name <- "Enter name: " 
            role <- "Enter role: " 
            age <- "Enter age: "
            nationality <- "Enter nationality: "

            astronauts[i] = NEW Astronaut(name, role, age, nationality)
        END FOR 

        RETURN astronauts
    END astronautArray 

    METHOD: addMissiondetails 
    IMPORT: filename (String), adventures (Mission[])
    EXPORT: none
    ASSERTION: Writes user input into the file using mission array.
    ALGORITHM: 
        FILEOUTPUTSTREAM fileStream = NULL

        fileStream = NEW FileOutputStream(fileName)
        pw = NEW PrintWriter(fileName)
        pw PRINT "Mission Name,Mission Code,Destination Planet,Launch Year,Success Rate,Manned Mission,Astronauts" 
        FOR MISSION mission : adventures
            IF mission NOT EQUAL NULL THEN 
                pw PRINT mission.toCSV() + "\n"
            END IF 
        END FOR 

        CLOSE pw
    END addMissiondetails

    METHOD: editMission
    IMPORT: pDatafile (String), missionArray (Mission[])
    EXPORT: none
    ASSERTION: Edit missions that already exist
    ALGORITHM: 
        missionToEdit <- ASK "What is the name of the mission you wish to edit? " 
        found = false 

        i = 0 
        matchIndex = -1 

        WHILE i <= missionArray.length - 1 AND matchIndex EQUALS -1 DO 
            IF missionArray[i].getMissionName EQUALS missionToEdit THEN
                PRINT "Found: " + "\n" + missionArray[i]

                newName <- ASK "Enter new mission name: "
                missionArray[i].setMissionName(newName)

                newCode <- ASK "Enter new mission code: "
                missionArray[i].setMissionCode(newCode)

                newDestination <- ASK "Enter new destination: " 
                missionArray[i].setDestinationPlanet(newDestination)

                newLaunchYear <- ASK "Enter new launch year: " 
                missionArray[i].setLaunchYear(newLaunchYear)

                newSuccessRate <- ASK "Enter new success rate: " 
                missionArray[i].setSuccessRate(newSuccessRate)

                newMannedStatus <- ASK "mission is manned (true or false): "
                missionArray[i].setMissionManned(newMannedStatus)

                MISSION mission = missionArray[i]
                    IF mission NOT EQUAL NULL THEN 
                        IF mission.getMissionManned AND mission.getAstronaut NOT EQUAL NULL THEN 
                            ASTRONAUT[] newAstronauts = astronautArray()
                            mission.setAstronauts(newAstronauts)
                        END IF 
                    END IF 
                missionArray[i] = mission

                found = true 
                matchIndex = i 
            END IF 
             i++
        END WHILE 

        IF FOUND EQUALS FALSE THEN
            PRINT "Cannot find Mission."
            RETURN 
        END IF 

        addMissiondetails(pDataFile, missionArray)
    END editMission

    METHOD: missionSummary
    IMPORT: missions (Mission[])
    EXPORT: none 
    ASSERTION: 
    ALGORITHM: 
        DOUBLE[] successRates = NEW double[missions.length]
        sum = 0.0 

        FOR i = 0 TO numOfMissions CHANGEBY i++
            IF mission[i] NOT EQUAL NULL THEN 
                successRate[i] = missions[i].getSuccessRate()
            END IF 
        END FOR 

        FOR j = 0 TO numberOfMissions CHANGEBY j++
            sum += successRate[j]
        END FOR 

        average = sum/numOFMissions
        PRINT "The average success rate is: " + average + "%" 
        DOUBLE[] sortedRates = sortedRates(successRate)
        PRINT "The highest success rate is: " + sortedRates[0] + "%"
        PRINT "The lowest success rate is: " + sortedRates[numOfMissions - 1] + "%"
    END missionSummary 

    METHOD: sortedRates
    IMPORT: successRates (double[])
    EXPORT: successRates (double[])
    ASSERTION: Selection sort largest to smallest
    ALGORITHM: 
        FOR i = 0 TO numOfMissions CHANGEBY i++
            maxIndex = i 
            FOR  j = (i + 1) TO numOfMissions CHANGEBY j++
                IF successRate[j] > successRates[maxIndex] THEN
                    maxIndex = j
                END IF 
            END FOR 
            temp = successRates[maxIndex]
            successRates[maxIndex] = successRates[i]
            successRates[i] = temp
        END FOR 
        RETURN successRates
    END sortedRates 

    METHOD: findNationality 
    IMPORT: missions (Mission[])
    EXPORT: none 
    ASSERTION: Show all astronauts for a given nationality 
    ALGORITHM:

        FOR i = 0 TO numOfMissions CHANGEBY i++

                    IF astronaut.getAstronautNationality EQUALS missionAstronaut 
                        PRINT astronaut + "\n"
                        found = true 
                    END IF 
                END FOR 
            END IF 
        END FOR 

        IF found EQUAL FALSE 
            PRINT "No astronaut found from: " + missionAstronaut
        END IF 
    END findNationality
END MissionController
